FastAPI & YOLOv8 Processing Pipeline

How the neural object detector parses document bounding boxes and feeds OCR tokenizers.

# Python FastAPI Backend (process_image & process_full_image)
from fastapi import FastAPI, UploadFile
from ultralytics import YOLO
import cv2
import numpy as np

app = FastAPI(title="Identity Document YOLOv8 Extraction Service")
model = YOLO("weights/best_id_detector.pt")  # Trained on ID cards & Passports

@app.post("/process_full_image")
async def extract_document_entities(file: UploadFile):
    # 1. Perspective alignment via OpenCV Homography
    image_bytes = await file.read()
    image = cv2.imdecode(np.frombuffer(image_bytes, np.uint8), cv2.IMREAD_COLOR)
    
    # 2. YOLOv8 Neural Inference for Bounding Boxes (Face, Name, DOB, ID, MRZ)
    results = model(image, conf=0.75)
    detections = []
    
    for r in results:
        for box in r.boxes:
            cls_name = model.names[int(box.cls)]
            coords = box.xyxy[0].tolist()  # [x1, y1, x2, y2]
            
            # Crop ROI and run OCR text recognition
            roi = image[int(coords[1]):int(coords[3]), int(coords[0]):int(coords[2])]
            extracted_text = run_tesseract_ocr(roi)
            detections.append({"class": cls_name, "box": coords, "text": extracted_text})
            
    return {"status": "SUCCESS", "entities": detections}
